cmake_minimum_required(VERSION 3.20)
project(BAKPoseProbeLauncher LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 required")
endif()
add_library(BAKPoseProbeLauncher SHARED ProbeLauncher.cpp)
target_compile_features(BAKPoseProbeLauncher PRIVATE cxx_std_20)
target_compile_definitions(BAKPoseProbeLauncher PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
set_target_properties(BAKPoseProbeLauncher PROPERTIES OUTPUT_NAME "BAKPoseProbeLauncher")
