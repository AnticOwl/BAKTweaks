cmake_minimum_required(VERSION 3.20)
project(BAKLightRefreshProbe LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 required")
endif()
add_subdirectory(../../third_party/minhook minhook)
add_library(BAKLightRefreshProbe SHARED Probe.cpp)
target_include_directories(BAKLightRefreshProbe PRIVATE ../../third_party/minhook/include)
target_compile_features(BAKLightRefreshProbe PRIVATE cxx_std_20)
target_compile_definitions(BAKLightRefreshProbe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(BAKLightRefreshProbe PRIVATE minhook)
add_library(BAKLightRefreshCleanup SHARED Probe.cpp)
target_include_directories(BAKLightRefreshCleanup PRIVATE ../../third_party/minhook/include)
target_compile_features(BAKLightRefreshCleanup PRIVATE cxx_std_20)
target_compile_definitions(BAKLightRefreshCleanup PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX PROBE_CLEANUP_ONLY)
target_link_libraries(BAKLightRefreshCleanup PRIVATE minhook)
